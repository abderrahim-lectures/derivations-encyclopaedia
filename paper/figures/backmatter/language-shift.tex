% German and Latin literature; the blue is English and the French works.

  \draw[->] (0,0) -- (16.4,0) node[right] {decade};

  \foreach \y in {0, 50, 100, 150, 200, 250, 300, 350, 400} {
    \draw[gray!40] (0,\y) -- (15,\y);
